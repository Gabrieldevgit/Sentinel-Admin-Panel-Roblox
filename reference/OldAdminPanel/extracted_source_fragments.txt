function startFlying()
	if isFlying then return end

	local char = player.Character
	if not char then 
		showNotification("
=====CHUNK=====
 Error", "Character not found", 2)
		return 
	end

	local hum = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	if not hum or not root then 
		showNotification("
=====CHUNK=====
 Error", "Humanoid/Root not found", 2)
		return 
	end

	isFlying = true

	-- Set humanoid state to flying (Anti-cheat bypass)
	hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Flying, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.GettingUp, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Landed, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Running, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, false)
	hum:ChangeState(Enum.HumanoidStateType.Flying)

	-- Create BodyGyro
	flyGyro = Instance.new("BodyGyro")
	flyGyro.P = 9000
	flyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyGyro.CFrame = root.CFrame
	flyGyro.Parent = root

	-- Create BodyVelocity
	flyVel = Instance.new("BodyVelocity")
	flyVel.Velocity = Vector3.new(0, 0, 0)
	flyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	flyVel.Parent = root

	-- Create Fly GUI
	createFlyGui()

	-- Fly loop
	flyConn = RunService.Heartbeat:Connect(function()
		if not isFlying then return end

		local c = player.Character
		if not c then stopFlying() return end

		local h = c:FindFirstChildOfClass("Humanoid")
		local r = c:FindFirstChild("HumanoidRootPart")
		if not h or not r then stopFlying() return end

		if not flyGyro or not flyVel then stopFlying() return end

		-- Keep flying state
		h:ChangeState(Enum.HumanoidStateType.Flying)

		local cam = workspace.CurrentCamera
		if not cam then return end

		local dir = Vector3.new(0, 0, 0)

		-- Movement direction
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then
			dir = dir + cam.CFrame.LookVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then
			dir = dir - cam.CFrame.LookVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then
			dir = dir - cam.CFrame.RightVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then
			dir = dir + cam.CFrame.RightVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
			dir = dir + Vector3.new(0, 1, 0)
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
			dir = dir + Vector3.new(0, -1, 0)
		end

		-- Speed multiplier
		local spd = flySpeed
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
			spd = spd * Config.FLY_SETTINGS.BOOST_MULTIPLIER
		end

		-- Apply movement
		if flyVel and flyVel.Parent then
			if dir.Magnitude > 0 then
				flyVel.Velocity = dir.Unit * spd
			else
				flyVel.Velocity = Vector3.new(0, 0, 0)
			end
		end

		if flyGyro and flyGyro.Parent then
			flyGyro.CFrame = cam.CFrame
		end
	end)

	showNotification("
=====CHUNK=====
 Fly Enabled", "WASD + Space/Shift | Ctrl = Boost", 3)
	print("
=====CHUNK=====
 Fly started")
end

function stopFlying()
	if not isFlying then return end
	isFlying = false

	if flyConn then
		flyConn:Disconnect()
		flyConn = nil
	end

	if flyGyro and flyGyro.Parent then
		flyGyro:Destroy()
	end
	flyGyro = nil

	if flyVel and flyVel.Parent then
		flyVel:Destroy()
	end
	flyVel = nil

	-- Destroy Fly GUI
	if flyGui then
		flyGui:Destroy()
		flyGui = nil
	end

	-- Re-enable humanoid states
	local char = player.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Flying, false)
			hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.GettingUp, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Landed, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Running, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
			hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
			hum:ChangeState(Enum.HumanoidStateType.Freefall)
		end
	end

	-- Reset noclip
	noclipEnabled = false
	remoteEvent:FireServer("ToggleNoclip", {})

	showNotification("
=====CHUNK=====
 Fly Disabled", "Stopped", 2)
	print("
=====CHUNK=====
 Fly stopped")
end

function executeCommand(cmdName)
	if not cmdName or type(cmdName) ~= "string" then return end

	local ts = os.date("[%H:%M:%S]")
	local selCount = 0
	for _ in pairs(selectedPlayers) do selCount = selCount + 1 end

	cmdName = cmdName:lower()

	-- Special commands that don't need targets
	if cmdName == "fly" then
		if isFlying then
			stopFlying()
		else
			startFlying()
		end
		table.insert(commandLogs, ts .. " Executed: fly")
		return

	elseif cmdName == "view" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select a player to view", 3) 
			return 
		end
		for targetName, _ in pairs(selectedPlayers) do
			startViewing(targetName)
			table.insert(commandLogs, ts .. " Executed: view " .. targetName)
			break
		end
		return

	elseif cmdName == "pm" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players to PM", 3) 
			return 
		end
		showPMPrompt(function(message)
			remoteEvent:FireServer("PrivateMessage", {
				message = message,
				targets = selectedPlayers
			})
			local names = {}
			for n, _ in pairs(selectedPlayers) do table.insert(names, n) end
			table.insert(commandLogs, ts .. " PM sent to: " .. table.concat(names, ", "))
			showNotification("
=====CHUNK=====
 PM Sent", "Message delivered", 2)
		end)
		return

	elseif cmdName == "bring" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3) 
			return 
		end
		remoteEvent:FireServer("Bring", {targets = selectedPlayers})
		table.insert(commandLogs, ts .. " Executed: bring")
		showNotification("
=====CHUNK=====
 Success", "Brought players", 2)
		return

	elseif cmdName == "goto" or cmdName == "to" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select a player", 3) 
			return 
		end
		remoteEvent:FireServer("Goto", {targets = selectedPlayers})
		table.insert(commandLogs, ts .. " Executed: goto")
		showNotification("
=====CHUNK=====
 Success", "Teleported to player", 2)
		return

	elseif cmdName == "fast" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3) 
			return 
		end
		remoteEvent:FireServer("Fast", {targets = selectedPlayers})
		table.insert(commandLogs, ts .. " Executed: fast (speed " .. Config.SPEED_PRESETS.FAST .. ")")
		showNotification("
=====CHUNK=====
 Success", "Fast speed applied", 2)
		return

	elseif cmdName == "slow" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3) 
			return 
		end
		remoteEvent:FireServer("Slow", {targets = selectedPlayers})
		table.insert(commandLogs, ts .. " Executed: slow (speed " .. Config.SPEED_PRESETS.SLOW .. ")")
		showNotification("
=====CHUNK=====
 Success", "Slow speed applied", 2)
		return

	elseif cmdName == "cleareffects" then

		table.insert(commandLogs, ts .. " Executed: cleareffects")

		showNotification("
=====CHUNK=====

		return

	elseif cmdName == "clearfog" then
		remoteEvent:FireServer("ClearFog", {})
		table.insert(commandLogs, ts .. " Executed: clearfog")
		showNotification("
=====CHUNK=====
 Success", "Fog cleared", 2)
		return

	elseif cmdName == "ambient" then
		remoteEvent:FireServer("Ambient", {})
		table.insert(commandLogs, ts .. " Executed: ambient")
		showNotification("
=====CHUNK=====
 Success", "Ambient set", 2)
		return

	elseif cmdName == "fixlighting" then
		remoteEvent:FireServer("FixLighting", {})
		table.insert(commandLogs, ts .. " Executed: fixlighting")
		showNotification("
=====CHUNK=====
 Success", "Lighting fixed", 2)
		return
	end

	-- KICK command with prompt
	if cmdName == "kick" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3) 
			return 
		end
		showKickPrompt(function(reason)
			remoteEvent:FireServer("ExecuteCommand", {
				command = "kick", 
				targets = selectedPlayers, 
				args = {reason = reason}
			})
			local names = {}
			for n, _ in pairs(selectedPlayers) do table.insert(names, n) end
			table.insert(commandLogs, ts .. " Executed: kick on " .. table.concat(names, ", ") .. " | Reason: " .. reason)
			showNotification("
=====CHUNK=====
 Success", "Kick executed", 2)
		end)
		return
	end

	-- BAN command with Perm/Temp prompt
	if cmdName == "ban" then
		if selCount == 0 then 
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3) 
			return 
		end
		showBanPrompt(function(banType, reason, duration)
			remoteEvent:FireServer("ExecuteBan", {
				banType = banType,
				targets = selectedPlayers,
				reason = reason,
				duration = duration
			})
			local names = {}
			for n, _ in pairs(selectedPlayers) do table.insert(names, n) end
			table.insert(commandLogs, ts .. " Executed: " .. banType .. " ban on " .. table.concat(names, ", ") .. " | Reason: " .. reason)
			showNotification("
=====CHUNK=====
 Success", banType .. " ban executed", 2)
		end)
		return
	end

	-- Check if command needs value
	local needsValue = false
	for _, cmd in ipairs(allCommands) do
		if cmd.name:lower() == cmdName and cmd.needsValue then
			needsValue = true
			break
		end
	end

	if needsValue then
		if selCount == 0 and cmdName ~= "time" and cmdName ~= "blur" and cmdName ~= "brightness" then
			showNotification("
=====CHUNK=====
 No Selection", "Select players", 3)
			return
		end

		showNumberPrompt(cmdName, function(value)
			remoteEvent:FireServer("ExecuteCommandWithValue", {
				command = cmdName,
				targets = selectedPlayers,
				value = value
			})
			table.insert(commandLogs, ts .. " Executed: " .. cmdName .. " (" .. value .. ")")
			showNotification("
=====CHUNK=====
 Success", cmdName .. " executed", 2)
		end)
		return
	end

	-- Regular commands
	if selCount == 0 then
		showNotification("
=====CHUNK=====
 No Selection", "Select players first", 3)
		return
	end

	remoteEvent:FireServer("ExecuteCommand", {
		command = cmdName,
		targets = selectedPlayers
	})

	table.insert(commandLogs, ts .. " Executed: " .. cmdName)
	showNotification("
=====CHUNK=====
 Success", cmdName .. " executed", 2)
end

		end

		stopViewing()
	end
end)

local function isGameOwner(player)
	return player and player.UserId == Config.GAME_OWNER_ID
end

local function getPlayerRank(player)
	if not player then return 0 end

	if isGameOwner(player) then
		return RANKS.Owner
	end

	for _, adminId in ipairs(Config.ADMIN_IDS) do
		if player.UserId == adminId then
			return RANKS.Admin
		end
	end

	for _, rankData in ipairs(adminRanks) do
		if rankData.userId == player.UserId then
			return RANKS[rankData.rank] or 0
		end
	end

	return 0
end

local function isAdmin(player)
	return getPlayerRank(player) >= RANKS.Mod
end

local function setPlayerRank(userId, username, rank)
	for i, rankData in ipairs(adminRanks) do
		if rankData.userId == userId then
			table.remove(adminRanks, i)
			break
		end
	end

	if rank ~= "None" then
		table.insert(adminRanks, {
			userId = userId,
			username = username,
			rank = rank,
			timestamp = os.date("%Y-%m-%d %H:%M:%S")
		})
	end

	saveAdminRanks()
	print("
=====CHUNK=====
 Set rank:", username, "->", rank)
end

local function getPlayerFromName(name)
	if not name or type(name) ~= "string" then return nil end

	local nameLower = name:lower()

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr and plr.Name and plr.Name:lower() == nameLower then
			return plr
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr and plr.Name and plr.Name:lower():find(nameLower) then
			return plr
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr and plr.DisplayName and plr.DisplayName:lower():find(nameLower) then
			return plr
		end
	end

	return nil
end

-- Notify target when command is executed on them
local function notifyTarget(target, adminName, commandName)
	if not target or not Config.NOTIFICATION_SETTINGS.SHOW_COMMAND_NOTIFICATIONS then return end

	pcall(function()
		remoteEvent:FireClient(target, "CommandNotification", {
			admin = adminName,
			command = commandName
		})
	end)
end

remoteEvent.OnServerEvent:Connect(function(player, action, data)
	if not player or not isAdmin(player) then 
		warn("
=====CHUNK=====
 Non-admin tried to use panel:", player and player.Name or "Unknown")
		return 
	end

	data = data or {}

	-- Execute Command
	if action == "ExecuteCommand" then
		local cmdName = data.command
		local targets = data.targets or {}
		local args = data.args or {}

		if not cmdName then return end

		local handler = commandHandlers[cmdName:lower()]
		if handler then
			for targetName, _ in pairs(targets) do
				local target = getPlayerFromName(targetName)
				if target then
					handler(target, player.Name)
				end
			end
		elseif cmdName:lower() == "kick" then
			local reason = args.reason or "Kicked by admin"
			for targetName, _ in pairs(targets) do
				local target = getPlayerFromName(targetName)
				if target then
					kickPlayer(target, reason, player.Name)
				end
			end
		elseif cmdName:lower() == "ban" then
			local reason = args.reason or "Banned by admin"
			for targetName, _ in pairs(targets) do
				local target = getPlayerFromName(targetName)
				if target then
					banPlayer(target, reason, player.Name)
				end
			end
		end

		-- Execute Command With Value
	elseif action == "ExecuteCommandWithValue" then
		local cmdName = data.command
		local targets = data.targets or {}
		local value = data.value

		if not cmdName or not value then return end

		local handler = commandsWithValue[cmdName:lower()]
		if handler then
			if cmdName:lower() == "time" or cmdName:lower() == "blur" or cmdName:lower() == "brightness" then
				handler(value)
			else
				for targetName, _ in pairs(targets) do
					local target = getPlayerFromName(targetName)
					if target then
						handler(target, value, player.Name)
					end
				end
			end
		end

		-- Bring Command
	elseif action == "Bring" then
		local targets = data.targets or {}
		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				bringPlayer(player, target)
			end
		end

		-- Goto Command
	elseif action == "Goto" then
		local targets = data.targets or {}
		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				gotoPlayer(player, target)
				break
			end
		end

		-- Fast/Slow Commands
	elseif action == "Fast" then
		local targets = data.targets or {}
		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				setSpeed(target, Config.SPEED_PRESETS.FAST, player.Name)
			end
		end

	elseif action == "Slow" then
		local targets = data.targets or {}
		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				setSpeed(target, Config.SPEED_PRESETS.SLOW, player.Name)
			end
		end

		-- View Command
	elseif action == "View" then
		local targetName = data.targetName
		if targetName then
			local target = getPlayerFromName(targetName)
			if target then
				print("
=====CHUNK=====
", player.Name, "viewing", target.Name)
			end
		end

		-- Fly Toggle
	elseif action == "ToggleFly" then
		print("
=====CHUNK=====
", player.Name, "toggled fly")

		-- Noclip Toggle
	elseif action == "ToggleNoclip" then
		noclipPlayers[player.Name] = not noclipPlayers[player.Name]
		print("
=====CHUNK=====
", player.Name, "noclip:", noclipPlayers[player.Name])

		-- Announcement
	elseif action == "Announce" then
		local message = data.message
		local scope = data.scope or "current"

		if not message then return end

		if scope == "current" then
			for _, plr in ipairs(Players:GetPlayers()) do
				remoteEvent:FireClient(plr, "Announcement", {
					message = message,
					from = player.Name
				})
			end
			print("
=====CHUNK=====
 Announcement sent to current server by", player.Name)

		elseif scope == "all" then
			pcall(function()
				MessagingService:PublishAsync("AdminAnnouncement", {
					message = message,
					from = player.Name,
					timestamp = os.time()
				})
			end)
			print("
=====CHUNK=====
 Global announcement sent by", player.Name)
		end

		-- Private Message
	elseif action == "PrivateMessage" then
		local message = data.message
		local targets = data.targets or {}

		if not message or #message == 0 then return end

		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				remoteEvent:FireClient(target, "PrivateMessage", {
					from = player.Name,
					message = message,
					timestamp = os.date("%H:%M:%S")
				})
				print("
=====CHUNK=====
 PM from", player.Name, "to", target.Name)
			end
		end

		-- PM Reply
	elseif action == "PMReply" then
		local targetName = data.targetName
		local message = data.message

		if not targetName or not message then return end

		local target = getPlayerFromName(targetName)
		if target then
			remoteEvent:FireClient(target, "PrivateMessage", {
				from = player.Name,
				message = message,
				timestamp = os.date("%H:%M:%S")
			})
			print("
=====CHUNK=====
 PM Reply from", player.Name, "to", target.Name)
		end

		-- Get Banned Players List
	elseif action == "GetBannedPlayers" then
		remoteEvent:FireClient(player, "BannedPlayersList", {bannedPlayers = bannedPlayers})

		-- Unban Player
	elseif action == "UnbanPlayer" then
		local userId = data.userId
		if userId then
			for i, banData in ipairs(bannedPlayers) do
				if banData.userId == userId then
					table.remove(bannedPlayers, i)
					saveBannedPlayers()
					print("
=====CHUNK=====
 Unbanned:", banData.username)
					break
				end
			end
		end

		-- Get Items List
	elseif action == "GetItems" then
		local items = {}
		local toolsFolder = ServerStorage:FindFirstChild("IngameTools")
		if toolsFolder then
			for _, item in ipairs(toolsFolder:GetChildren()) do
				if item:IsA("Tool") then
					table.insert(items, item.Name)
				end
			end
		end
		remoteEvent:FireClient(player, "ItemsList", {items = items})

		-- Give Item
	elseif action == "GiveItem" then
		local itemName = data.itemName
		local targets = data.targets or {}

		if not itemName then return end

		for targetName, _ in pairs(targets) do
			local target = getPlayerFromName(targetName)
			if target then
				giveItem(target, itemName, player.Name)
			end
		end

		-- Get Admin List (For Rank Management)
	elseif action == "GetAdminList" then
		if not isGameOwner(player) then
			remoteEvent:FireClient(player, "NotGameOwner", {})
			return
		end

		remoteEvent:FireClient(player, "AdminList", {admins = adminRanks})

		-- Set Admin Rank
	elseif action == "SetAdminRank" then
		if not isGameOwner(player) then
			remoteEvent:FireClient(player, "NotGameOwner", {})
			return
		end

		local userId = data.userId
		local username = data.username
		local rank = data.rank

		if userId and username and rank then
			setPlayerRank(userId, username, rank)
			print("
=====CHUNK=====
", player.Name, "set", username, "to", rank)
		end

		-- World Commands
	elseif action == "ClearEffects" then
		clearEffects()

	elseif action == "Disco" then
		toggleDisco()

	elseif action == "Day" then
		setDay()

	elseif action == "Night" then
		setNight()

	elseif action == "Sunset" then
		setSunset()

	elseif action == "Dawn" then
		setDawn()

	elseif action == "Foggy" then
		setFoggy()

	elseif action == "ClearFog" then
		clearFog()

	elseif action == "Ambient" then
		setAmbient()

	elseif action == "FixLighting" then
		fixLighting()
	end
end)

Players.PlayerAdded:Connect(function(player)
	-- Check if banned
	for _, banData in ipairs(bannedPlayers) do
		if banData.userId == player.UserId then
			player:Kick("You are banned!\nReason: " .. (banData.reason or "No reason provided") .. "\nBanned by: " .. (banData.bannedBy or "Unknown"))
			return
		end
	end

	-- Check server lock
	if serverLocked and not isAdmin(player) then
		player:Kick("Server is locked!\nOnly admins can join.")
		return
	end

	print("
=====CHUNK=====
", player.Name, "joined")

	-- Send current rank to player if admin
	if isAdmin(player) then
		task.wait(2)

		local rankName = "None"
		local rankLevel = getPlayerRank(player)

		if isGameOwner(player) then
			rankName = "Owner"
		elseif rankLevel >= RANKS.HeadAdmin then
			rankName = "HeadAdmin"
		elseif rankLevel >= RANKS.Admin then
			rankName = "Admin"
		elseif rankLevel >= RANKS.Mod then
			rankName = "Mod"
		end

		remoteEvent:FireClient(player, "CurrentRank", {
			rank = rankName,
			isGameOwner = isGameOwner(player)
		})

		print("
=====CHUNK=====
 Sent rank to", player.Name, ":", rankName)
	end

	-- Keep jailed players in jail
	player.CharacterAdded:Connect(function(char)
		task.wait(0.5)
		if jailedPlayers[player.Name] then
			local jail = jailedPlayers[player.Name]
			if jail and jail.Parent then
				local rootPart = getRootPart(player)
				if rootPart then
					rootPart.CFrame = jail.CFrame
				end
			end
		end

		-- Maintain god mode
		if godMode[player.Name] then
			task.wait(1)
			godPlayer(player, "System")
		end

		-- Restart loop kill
		if loopKilling[player.Name] then
			task.wait(1)
			task.spawn(function()
				while loopKilling[player.Name] do
					killPlayer(player, "System")
					task.wait(2)
				end
			end)
		end
	end)
end)

	-- Clean up loop kills
	loopKilling[player.Name] = nil
	godMode[player.Name] = nil
	mutedPlayers[player.Name] = nil
	noclipPlayers[player.Name] = nil

	print("
=====CHUNK=====
", player.Name, "left")
end)

RunService.Stepped:Connect(function()
	for playerName, enabled in pairs(noclipPlayers) do
		if enabled then
			local player = Players:FindFirstChild(playerName)
			if player then
				local char = getCharacter(player)
				if char then
					for _, part in ipairs(char:GetDescendants()) do

							part.CanCollide = false
						end
					end
				end
			end
		end
	end

RunService.Heartbeat:Connect(function()
	for playerName, jail in pairs(jailedPlayers) do
		if jail and jail.Parent then
			local player = Players:FindFirstChild(playerName)
			if player then
				local rootPart = getRootPart(player)
				if rootPart then
					local distance = (rootPart.Position - jail.Position).Magnitude
					if distance > 6 then
						rootPart.CFrame = jail.CFrame
					end
				end
			end
		end
	end
end)
